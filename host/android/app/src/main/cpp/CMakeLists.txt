cmake_minimum_required(VERSION 3.22.1)
project(littlelegendhost)
set(NODE_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../nodejs-mobile)
if(NOT EXISTS ${NODE_ROOT}/include/node/node.h)
 message(FATAL_ERROR "Place nodejs-mobile Android headers under src/main/nodejs-mobile/include/node and libnode.so under jniLibs/arm64-v8a first")
endif()
add_library(node SHARED IMPORTED)
set_target_properties(node PROPERTIES IMPORTED_LOCATION ${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}/libnode.so)
add_library(legendhost SHARED native-host.cpp)
target_include_directories(legendhost PRIVATE ${NODE_ROOT}/include/node)
target_link_libraries(legendhost node log)
